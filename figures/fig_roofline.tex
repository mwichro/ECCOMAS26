% Roofline of the Laplace operator on the A100 80GB SXM (FP64).
% TODO(benchmark): fill in measured (AI, TFLOP/s) points for p = 1..7, best (B=16-32) config.
\begin{tikzpicture}
    \begin{loglogaxis}[
            width=\linewidth,
            height=0.85\linewidth,
            xlabel={Arithmetic intensity [FLOP/byte]},
            ylabel={Performance [TFLOP/s]},
            log basis x=2,
            xmin=0.5, xmax=16,
            xtick={0.5,1,2,4,8,16},
            xticklabels={$\frac{1}{2}$,1,2,4,8,16},
            log basis y=2,
            ymin=1, ymax=32,
            ytick={1,2,4,8,16,32},
            yticklabels={1,2,4,8,16,32},
            grid=both,
            legend pos=south east,
            legend cell align=left,
        ]
        % --- Colored regions (drawn first so rooflines appear on top) ---
        % Green: CUDA core attainable region — below the vector FP64 roofline
        \addplot[green, fill=green, draw=none, fill opacity=0.3, forget plot]
        coordinates {(0.5,1) (4.76,9.7) (16,9.7) (16,1) (0.5,1)};
        % Blue: tensor core extra region — above CUDA core ceiling up to tensor core ceiling.
        % Follows the bandwidth slope from CUDA ridge (4.76,9.7) to tensor ridge (9.56,19.5), then flat to x=16.
        \addplot[cyan, fill=cyan, draw=none, fill opacity=0.3, forget plot]
        coordinates {(4.76,9.7) (9.56,19.5) (16,19.5) (16,9.7) (4.76,9.7)};

        % --- Roofline for the A100 80GB SXM (FP64): bandwidth 2039 GB/s ---
        % Vector FP64 peak: 9.7 TFLOP/s,  ridge at 9.7/2.039  ~ 4.76 FLOP/byte.
        % Tensor FP64 peak: 19.5 TFLOP/s, ridge at 19.5/2.039 ~ 9.56 FLOP/byte.
        \addplot[thick, domain=0.5:9.56, samples=2, forget plot] {2.039*x}
        node[pos=0.35, sloped, above, font=\small\bfseries] {Bandwidth 2039 GB/s};
        \addplot[thick, domain=0.5:16, samples=2, forget plot] {9.7}
        node[pos=0.0, above right, font=\small\bfseries] {CUDA 9.7 TFLOP/s};
        \addplot[thick, dashed, domain=0.5:16, samples=2, forget plot] {19.5}
        node[pos=0.0, above right, font=\small\bfseries] {Tensor 19.5 TFLOP/s};

        % --- Measured operator points ---
        % Laplace: from tab:perf_teaser_laplace
        \addplot[black, mark=*, mark size=2pt]
        table[col sep=comma, x=ai, y=tflops] {raw_results/laplace_roofline.csv};
        \addlegendentry{Laplace}
        % Mass (volumetric): from raw_results/dss_struct, volumetric_p* rows
        \addplot[red, mark=square, mark options={fill=none}, mark size=2pt]
        table[col sep=comma, x=ai, y=tflops] {raw_results/mass_roofline.csv};
        \addlegendentry{Mass}
    \end{loglogaxis}
\end{tikzpicture}
