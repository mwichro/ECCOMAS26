\begin{frame}{Setting: high-order FEM + geometric multigrid}
    \begin{itemize}
        \item Poisson-type problems, $\mathbb Q_p$ tensor-product elements, mesh hierarchy $\mesh_1 \sqsubset \dots \sqsubset
                  \mesh_L$
        \item Matrix-free operator evaluation with sum factorization: $\mathcal O(p^{d+1})$ work, $\mathcal O(p^d)$ memory
        \item Multigrid V-cycle needs a \alert{smoother} on every level
    \end{itemize}
    \vfill
    \begin{block}{Vertex patch (Schwarz) smoothers}
        \begin{itemize}
            \item Overlapping subspace correction on patches of $2^d$ cells around each vertex
            \item Robust in the polynomial degree $p$ \hfill {\small [Pavarino '93]}
            \item Excellent data locality --- great fit for modern hardware
            \item Handle Stokes, CutFEM, shifted boundary, \dots
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{The classical answer: fast diagonalization}
    For a \alert{separable} operator on a \alert{Cartesian} patch (Lynch, Rice, Thomas '64):
    \begin{gather*}
        A_j = L_x \otimes M_y + M_x \otimes L_y
        \qquad\Longrightarrow\qquad
        A_j^{-1} = (Q_x \otimes Q_y)\, \Lambda^{-1} (Q_x \otimes Q_y)^T
    \end{gather*}
    with 1D eigendecompositions $L Q = M Q \Lambda$.
    \begin{itemize}
        \item Exact inverse, applied with tensor contractions: $\mathcal O(p^{d+1})$ --- same cost as one operator application
        \item The workhorse of state-of-the-art patch smoothers \hfill {\small [Witte, Arndt, Kanschat '21]}
    \end{itemize}
    \vfill
    \begin{alertblock}{But\dots}
        \begin{itemize}
            \item Requires separability: Cartesian cells, constant coefficients
            \item Deteriorates on deformed grids, breaks on unstructured patches
        \end{itemize}
    \end{alertblock}
    \vfill
    \centering
    So what do we do on a \emph{general} patch?
\end{frame}

\begin{frame}{The box we are already in: multigrid}
    How was multigrid invented in the first place?
    \begin{enumerate}
        \item Jacobi/Gauss--Seidel kill high frequencies, stall on low ones
        \item Fix: correct the low frequencies on a coarser grid $\to$ two-level method
        \item \emph{``And how do we solve the coarse problem?''} \quad
              \alert{Add another level.} Recurse.
    \end{enumerate}
    \vfill
    \begin{block}{The local patch problem is \emph{the same question}}
        \emph{``How do we solve the local problem?''} \quad \alert{Add more levels!}
        \begin{itemize}
            \item Solve each patch problem with its own multigrid
            \item h-coarsening impossible: a patch is only $2^d$ cells
            \item $\Rightarrow$ coarsen in the polynomial degree: \alert{p-multigrid}
        \end{itemize}
    \end{block}
    \vfill
    \centering \emph{Multigrid-within-multigrid:} stay matrix-free, iterative, and lean --- all the way down.
\end{frame}

\begin{frame}{The local p-multigrid solver}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Hierarchy on the patch} (mesh fixed):
            \begin{gather*}
                V_{j,1} \subset V_{j,3} \subset V_{j,7} \subset \dots \subset V_{j,p}
            \end{gather*}
            nested degrees $p_{k+1} = 2p_k + 1$
            \begin{itemize}
                \item Transfers: tensorized $L^2$ projections,\\ $\mathcal O(p^{d+1})$ via sum factorization
                \item Level operators: matrix-free,\\ evaluated on lower-degree basis
                \item Smoother: damped Jacobi\\ (or fast diagonalization of the\\ \emph{reference} patch as preconditioner:\\ the
                      ``Cartesian-reinforced'' smoother)
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{block}{Coarse solver ($p=1$)}
                One interior node left:
                \begin{gather*}
                    A_{j,1}^{-1} = \frac{1}{A_{j,1}}
                \end{gather*}
                The coarse solve is \alert{a single division}.
            \end{block}
            \vspace{0.5em}
            \begin{itemize}
                \item Memory: $\mathcal O(p^d)$ --- optimal
                \item Work per V-cycle: $\mathcal O(p^{d+1})$,\\ same as \emph{one} operator application (up to a constant)
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Rewrite assumptions: what modern hardware wants}
    \textbf{Target: GPUs} (A100: 2\,TB/s DRAM, fp64 tensor cores)
    \begin{itemize}
        \item Nominal bandwidth attainable \alert{only} with \emph{coalesced} access: a warp must read consecutive words
        \item An uncoalesced gather fetches a whole cache line per useful word
    \end{itemize}
    \vfill
    \begin{block}{The classical matrix-free pipeline is built around its worst operation}
        gather cell DoFs $\to$ compute $\to$ scatter-accumulate (atomics!)
        \\ Indirect addressing on \emph{every} operator application.
    \end{block}
    \vfill
    \begin{block}{My answer: make the \emph{cell-wise} vector the persistent storage}
        \begin{itemize}
            \item DoFs duplicated across element interfaces --- DG-like storage for CG
            \item Every access coalesced; no read-accumulate, no atomics
            \item Price: up to $8\times$ memory at $p=1$, shrinking fast with $p$
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{The key identity: primal--dual pairing}
    Two kinds of vectors live in the redundant cell-wise space $\mathbb V_{\text{cell}}$:
    \begin{itemize}
        \small
        \item \textbf{Primal} $u$: continuous fields, duplicated entries agree: $u = Cu$\\
              {\footnotesize ($C$ = any idempotent projection onto continuous fields; encodes hanging nodes too)}
        \item \textbf{Dual} $\tilde r$: raw unassembled cell integrals (residuals) --- valid \emph{without} summation
    \end{itemize}
    \begin{block}{Duality pairing identity}
        \vspace{-1.2em}
        \begin{gather*}
            u = Cu \quad\Longrightarrow\quad
            \langle \tilde r, u \rangle
            = \langle \tilde r, C u \rangle
            = \langle C^T \tilde r, u \rangle
        \end{gather*}
        \vspace{-1.6em}\\
        {\small The \emph{local} pairing of unassembled residuals with a continuous field
        \alert{equals the assembled global pairing.}}
    \end{block}
    \begin{itemize}
        \small
        \item CG/FGMRES written so every inner product pairs one primal with one dual $\Rightarrow$ \alert{no assembly, no
                  communication in the Krylov loop}; provably identical iterates to the classical assembled solver
        \item Only the preconditioner bridges the spaces (one DSS inside it)
    \end{itemize}
\end{frame}

\begin{frame}{Direct stiffness summation without indirection}
    \begin{center}
        \resizebox{0.98\textwidth}{!}{\input{figures/fig_dss_propagation.tex}}
    \end{center}
    \vspace{0.5em}
    \begin{itemize}
        \item No atomics, no coloring, no index lists. Every access coalesced

\begin{frame}{Multigrid on top: hanging nodes for free}
    Geometric multigrid on adaptive meshes, cell-wise storage:
    \begin{itemize}
        \small
        \item Classically: explicit hanging-node constraint matrix $C_{HN}$, edge operators, special transfer handling [Janssen \&
                      Kanschat '11]
        \item Cell-wise: the constraint is \emph{absorbed} into the primal--dual structure --- $C_{HN} \mathcal P = \mathcal P$, so
              it \alert{resolves into an identity} in the transfer
        \item Level-wise masking + DSS is all that remains; \alert{DSS across hanging nodes: never needed}
    \end{itemize}
    \vfill

\begin{frame}{Implementation: Triton}
    The \emph{entire} solver stack is written in \alert{Triton}
    \begin{itemize}
        \item Python compiled directly to assembly on NVIDIA \emph{and} AMD
        \item You write tile-level operations; the compiler handles registers, shared memory, coalescing
        \item Sum-factorized 1D contractions map onto tensor-core (MMA) tiles
    \end{itemize}
    \vfill
    \begin{block}{One catch: fp64 tensor cores want $8\times8\times4$ tiles}
        For moderate $p$ the 1D matrices are too small for the tile shapes Triton supported.
        \\[0.4em]
        $\Rightarrow$ I implemented the small-tile fp64 MMAv2 shape ($8{\times}8{\times}4$) in the Triton
        compiler and upstreamed it (\texttt{triton-lang/triton \#10060}).
        \\[0.4em]
        On the benchmarked MMA shapes, Triton now \alert{sometimes outperforms cuBLAS}.
    \end{block}
    \vfill
    None of the results on the previous slides are possible without this.
\end{frame}

\begin{frame}{Conclusions}
    \begin{itemize}
        \item Patch smoothers are worth it --- if the local solve is cheap. \alert{Multigrid-within-multigrid} makes it cheap,
              matrix-free, and robust (p- and h-robust, distorted meshes, coefficient jumps)
        \item On modern hardware, \alert{bytes beat FLOPs}: iterative local solvers outrun direct ones, and cell-wise storage outruns
              assembled pipelines
              \begin{itemize}
                  \item primal--dual pairing identity $\Rightarrow$ communication-free Krylov
                  \item hanging nodes resolve into identity constraints
              \end{itemize}
        \item Rewriting from scratch (in Triton) was the right call
    \end{itemize}
